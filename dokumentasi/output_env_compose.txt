=== ISI docker-compose.yml ===
name: proyek1-basisdata

services:
  postgres:
    image: postgres:15
    container_name: postgres_${NAMA}
    restart: unless-stopped
    environment:
      # Tidak ada nilai hardcode: semua dibaca dari file .env
      POSTGRES_USER: ${POSTGRES_USER}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_DB: ${POSTGRES_DB}
      BACKEND_DEV_PASSWORD: ${BACKEND_DEV_PASSWORD}
      BI_DEV_PASSWORD: ${BI_DEV_PASSWORD}
      DATA_ENGINEER_PASSWORD: ${DATA_ENGINEER_PASSWORD}
    ports:
      - "127.0.0.1:${PG_PORT}:5432"   # eksternal 22086 -> internal 5432
    volumes:
      - pg_data:/var/lib/postgresql/data
      - ./sql:/sql:ro
      - ./tests:/tests:ro
    networks:
      - net_mbd
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}"]
      interval: 5s
      timeout: 5s
      retries: 10

  pgadmin:
    image: dpage/pgadmin4:latest
    container_name: pgadmin_${NAMA}
    restart: unless-stopped
    environment:
      PGADMIN_DEFAULT_EMAIL: ${PGADMIN_DEFAULT_EMAIL}
      PGADMIN_DEFAULT_PASSWORD: ${PGADMIN_DEFAULT_PASSWORD}
    ports:
      - "127.0.0.1:${PGADMIN_PORT}:80"   # eksternal 44086 -> internal 80
    volumes:
      - pgadmin_data:/var/lib/pgadmin
    networks:
      - net_mbd
    depends_on:
      postgres:
        condition: service_healthy

volumes:
  pg_data:
  pgadmin_data:

networks:
  net_mbd:
    driver: bridge
